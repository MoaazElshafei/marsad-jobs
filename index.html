async function fetchAndDisplayData() { const container = document.getElementById('jobs-container'); // المكان الذي ستعرض فيه الوظائف try { // 1. ضع رابط الـ API الذي تريد ربطه هنا const apiUrl = 'https://api.example.com/jobs'; const response = await fetch(apiUrl, { method: 'GET', headers: { 'Content-Type': 'application/json', // إذا كان الـ API يحتاج مفتاح سري (119f5c4954msh962fe5958465253p1c7957jsne1c2b71ebd3b)، أضفه هنا: // 'Authorization': 'Bearer YOUR_API_KEY' } }); const data = await response.json(); // 2. التحقق من وجود بيانات if (!data || data.length === 0) { container.innerHTML = '

لا توجد بيانات متاحة حالياً.

'; return; } // 3. تحويل البيانات وعرضها داخل بطاقات HTML container.innerHTML = data.map(item => `

${item.title}

${item.company} - ${item.location}

التفاصيل ↗
`).join(''); } catch (error) { console.error('حدث خطأ أثناء الاتصال بالـ API:', error); container.innerHTML = '

فشل تحميل البيانات، يرجى المحاولة لاحقاً.

'; } } // تشغيل الدالة بمجرد فتح الصفحة fetchAndDisplayData();
